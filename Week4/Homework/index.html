let history = []; 
const maxHistory = 100; 
function setup() {
  createCanvas(600, 600);
  strokeWeight(2);
}
function draw() {
  background(0, 10);
  let noiseInput = frameCount * 0.01;
  let x = map(noise(noiseInput), 0, 1, 0, width);
  let y = map(noise(noiseInput + 0.5), 0, 1, 0, height);
  history.push(createVector(x, y)); 
  if (history.length > maxHistory) {
    history.shift(); // Remove the oldest point
  }
  beginShape();
  for (let p of history) {
    vertex(p.x, p.y);
  }
  endShape();
}
